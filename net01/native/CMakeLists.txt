cmake_minimum_required(VERSION 3.16)
project(net01_payload_transport LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT NS3_DIR OR NOT CONFIG_HEADER_DIR)
  message(FATAL_ERROR "Explicit existing NS3_DIR and generated CONFIG_HEADER_DIR required")
endif()
add_executable(net01-payload payload_transport.cc)
find_package(Threads REQUIRED)
target_link_libraries(net01-payload PRIVATE Threads::Threads)
target_include_directories(net01-payload PRIVATE "${NS3_DIR}/build/include" "${CONFIG_HEADER_DIR}")
foreach(module wifi internet csma mobility propagation spectrum antenna energy network core traffic-control bridge)
  find_library(ns3_${module} NAMES ns3.40-${module}-default PATHS "${NS3_DIR}/build/lib" NO_DEFAULT_PATH REQUIRED)
  target_link_libraries(net01-payload PRIVATE "${ns3_${module}}")
endforeach()
set_target_properties(net01-payload PROPERTIES BUILD_RPATH "${NS3_DIR}/build/lib")
