cmake_minimum_required(VERSION 3.16)
project(net01_time_control LANGUAGES CXX)
find_package(ament_cmake REQUIRED)
find_package(gazebo 11 REQUIRED)
find_package(gazebo_ros REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_srvs REQUIRED)
add_library(net01_time_control SHARED time_control_plugin.cc)
target_compile_features(net01_time_control PRIVATE cxx_std_17)
target_include_directories(net01_time_control PRIVATE ${GAZEBO_INCLUDE_DIRS})
target_link_directories(net01_time_control PRIVATE ${GAZEBO_LIBRARY_DIRS})
ament_target_dependencies(net01_time_control gazebo_ros rclcpp std_srvs)
target_link_libraries(net01_time_control ${GAZEBO_LIBRARIES})
