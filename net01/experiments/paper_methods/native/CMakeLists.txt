cmake_minimum_required(VERSION 3.16)
project(paper_methods LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT NS3_BUILD_DIR OR NOT CONFIG_HEADER_DIR)
  message(FATAL_ERROR "Explicit existing matching ns3 build and V3 header required")
endif()
add_executable(paper-methods paper_methods.cc)
target_include_directories(paper-methods PRIVATE "${NS3_BUILD_DIR}/include" "${CONFIG_HEADER_DIR}")
foreach(module wifi internet csma mobility propagation spectrum antenna energy network core traffic-control bridge)
  find_library(ns3_${module} NAMES ns3.40-${module} PATHS "${NS3_BUILD_DIR}/lib" NO_DEFAULT_PATH REQUIRED)
  target_link_libraries(paper-methods PRIVATE "${ns3_${module}}")
endforeach()
set_target_properties(paper-methods PROPERTIES BUILD_RPATH "${NS3_BUILD_DIR}/lib")
